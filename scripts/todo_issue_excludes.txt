# Paths skipped by scripts/check_todo_issues.py.
# Patterns are relative to the directory being scanned. Blank lines and
# full-line comments (#) are ignored.
#
#   path/to/dir/       directory prefix from the scan root
#   **/dirname/        a directory with that name at any depth
#   **/filename        a file with that name at any depth
#   **/*.ext           glob matched against the file name (* does not need to span /)
#
# Markdown and other non-source files are not scanned. The directory prefixes
# below still matter: learning pages and plans include source files, and the
# prefixes keep task-shaped text out of the check if the source list grows.
